# P3.2 扩展：通用 Python/Shell 语言框架 + Claude CLI 安装脚本生成

## Context

用户希望基于 DevPalAgent 的 P3.2 多语言框架，先建立通用的 Python 和 Shell 脚本语言支持，然后将 Claude CLI 安装脚本作为这个框架的应用实例。

**核心思路转变**：
- ❌ 旧思路：直接为安装脚本创建专用模板
- ✅ 新思路：先完善 P3.2 多语言框架（Python + Shell），安装脚本是框架的应用案例

**背景**：
- P3.2 原计划：支持 Python 项目生成（pytest + pyproject.toml）
- 现有状态：仅支持 C++，Python 插件未实现
- 新需求：同时支持 Shell 脚本（Bash/Batch）作为独立语言类型

# devpal/core/templates/shell_templates.py（新增）

@registry.register
class ShellProjectSkeletonTemplate(BaseTemplate):
    """Shell 脚本项目骨架模板"""
    name = "shell_project_skeleton"
    category = TemplateCategory.CORE
    language = "shell"
    priority = 100
    
    def generate(self, context: TemplateContext) -> List[GeneratedFile]:
        files = []
        
        # 生成 Bash 脚本
        if "bash" in context.variables.get("script_types", ["bash"]):
            files.append(GeneratedFile(
                path=f"scripts/{context.project_name}.sh",
                content=self._generate_bash_skeleton(context),
                description="Bash 脚本骨架"
            ))
        
        # 生成 Batch 脚本
        if "batch" in context.variables.get("script_types", []):
            files.append(GeneratedFile(
                path=f"scripts/{context.project_name}.bat",
                content=self._generate_batch_skeleton(context),
                description="Batch 脚本骨架"
            ))
        
        # 生成库函数
        files.append(GeneratedFile(
            path="lib/common.sh",
            content=self._generate_common_lib(),
            description="通用库函数"
        ))
        
        return files
    
    def _generate_bash_skeleton(self, context: TemplateContext) -> str:
        return f'''#!/bin/bash
# {context.project_name}
# Generated by DevPalAgent

set -e

# Source common library
SCRIPT_DIR="$(cd "$(dirname "${{BASH_SOURCE[0]}}")" && pwd)"
source "$SCRIPT_DIR/../lib/common.sh"

# Main function
main() {{
    log_info "Starting {context.project_name}"
    
    # TODO: Implement main logic
    
    log_info "Completed successfully"
}}

main "$@"
'''
    
    def _generate_batch_skeleton(self, context: TemplateContext) -> str:
        return f'''@echo off
REM {context.project_name}
REM Generated by DevPalAgent

setlocal enabledelayedexpansion

REM Main logic
:main
echo Starting {context.project_name}

REM TODO: Implement main logic

echo Completed successfully
exit /b 0
'''
    
    def _generate_common_lib(self) -> str:
        return '''#!/bin/bash
# Common library functions

log_info() {
    echo "[INFO] $1"
}

log_error() {
    echo "[ERROR] $1" >&2
}

check_command() {
    command -v "$1" >/dev/null 2>&1
}
'''

# devpal/core/i18n/base.py
class Locale(Enum):
    EN = "en"
    ZH = "zh"
    JA = "ja"
    KO = "ko"

class MessageCatalog(ABC):
    @abstractmethod
    def get_messages(self) -> Dict[str, str]:
        pass

class I18nContext:
    def __init__(self, locale: Locale = Locale.EN):
        self.locale = locale
        self._catalogs: Dict[Locale, MessageCatalog] = {}
    
    def register_catalog(self, locale: Locale, catalog: MessageCatalog):
        self._catalogs[locale] = catalog
    
    def t(self, key: str, **kwargs) -> str:
        """翻译函数"""
        catalog = self._catalogs.get(self.locale)
        if not catalog:
            return key
        message = catalog.get(key)
        if kwargs:
            try:
                message = message.format(**kwargs)
            except KeyError:
                pass
        return message

# devpal/core/i18n/locales/en.py
class EnglishMessages(MessageCatalog):
    def get_messages(self) -> dict:
        return {
            # Common messages
            "common.welcome": "Welcome to {app_name}",

### 第四层：Claude CLI 安装脚本（应用实例）

```python
# devpal/core/templates/installer_templates.py（新增）

@registry.register
class ClaudeCliInstallerTemplate(BaseTemplate):
    """Claude CLI 安装脚本模板（基于 Shell 语言框架）"""
    name = "claude_cli_installer"
    category = TemplateCategory.INSTALL
    language = "shell"
    priority = 100
    
    def should_apply(self, context: TemplateContext) -> bool:
        return "claude_cli_installer" in context.features
    
    def generate(self, context: TemplateContext) -> List[GeneratedFile]:
        # 使用 i18n 系统
        i18n = self._setup_i18n(context)
        
        files = []
        
        # 生成 Bash 脚本
        files.append(GeneratedFile(
            path="scripts/install_claude_cli.sh",
            content=self._generate_bash_installer(context, i18n),
            description="Claude CLI Bash installer"
        ))
        
        # 生成 Batch 脚本
        files.append(GeneratedFile(
            path="scripts/install_claude_cli.bat",
            content=self._generate_batch_installer(context, i18n),
            description="Claude CLI Batch installer"
        ))
        
        # 生成 Python 跨平台脚本
        files.append(GeneratedFile(
            path="scripts/install_claude_cli.py",
            content=self._generate_python_installer(context, i18n),
            description="Claude CLI Python installer"
        ))
        
        return files
    
    def _setup_i18n(self, context: TemplateContext) -> I18nContext:
        from devpal.core.i18n.base import I18nContext, Locale
        from devpal.core.i18n.locales.en import EnglishMessages
        from devpal.core.i18n.locales.zh import ChineseMessages
        
        i18n = I18nContext(Locale.EN)
        i18n.register_catalog(Locale.EN, EnglishMessages())
        i18n.register_catalog(Locale.ZH, ChineseMessages())
        return i18n
    
    def _generate_bash_installer(self, context: TemplateContext, i18n: I18nContext) -> str:
        # 收集所有语言的消息
        messages = self._collect_messages(i18n, ["en", "zh"])
        
        # 生成消息查找函数
        msg_function = self._generate_bash_msg_function(messages)
        
        return f'''#!/bin/bash
# Claude Code CLI Installer
# Generated by DevPalAgent

set -e

SCRIPT_LANG="en"

detect_language() {{
    for arg in "$@"; do
        if [[ "$arg" == "--lang=zh" ]]; then
            SCRIPT_LANG="zh"
            return
        elif [[ "$arg" == "--lang=en" ]]; then
            SCRIPT_LANG="en"
            return
        fi
    done
    
    if [[ "$LANG" =~ ^zh ]]; then
        SCRIPT_LANG="zh"
    fi
}}

{msg_function}

check_command() {{
    command -v "$1" >/dev/null 2>&1
}}

main() {{
    detect_language "$@"
    
    echo "$(msg 'install.welcome' 'Claude Code CLI')"
    echo ""
    
    # Check Node.js
    echo "$(msg 'install.check_node')"
    if ! check_command node; then
        echo "$(msg 'install.node_missing')"
        exit 1
    fi
    
    NODE_VERSION=$(node --version | sed 's/v//')
    echo "$(msg 'install.node_found' "$NODE_VERSION")"
    
    # Install Claude CLI
    echo "$(msg 'install.installing_cli')"
    if npm install -g @anthropic-ai/claude-code; then
        echo "$(msg 'common.success')"
    else
        echo "$(msg 'common.failed' 'npm install failed')"
        exit 1
    fi
    
    echo ""
    echo "$(msg 'install.done')"
}}

main "$@"
'''
    
    def _generate_python_installer(self, context: TemplateContext, i18n: I18nContext) -> str:
        """生成 Python 跨平台安装脚本"""
        return '''#!/usr/bin/env python3
"""
Claude Code CLI Installer (Cross-platform)
Generated by DevPalAgent
"""

import os
import sys
import subprocess
import platform
from typing import Dict

class Installer:
    def __init__(self, lang: str = "en"):
        self.lang = lang
        self.messages = self._load_messages()
    
    def _load_messages(self) -> Dict[str, Dict[str, str]]:
        return {
            "en": {
                "welcome": "Welcome to Claude Code CLI Installer",
                "check_node": "Checking Node.js installation...",
                "node_found": "✓ Node.js found: {version}",
                "success": "✓ Installation successful!",
            },
            "zh": {
                "welcome": "欢迎使用 Claude Code CLI 安装程序",
                "check_node": "检查 Node.js 安装状态...",
                "node_found": "✓ 已找到 Node.js: {version}",
                "success": "✓ 安装成功！",
            }
        }
    
    def msg(self, key: str, **kwargs) -> str:
        text = self.messages.get(self.lang, {}).get(key, key)
        return text.format(**kwargs) if kwargs else text
    
    def check_command(self, cmd: str) -> bool:
        try:
            subprocess.run([cmd, "--version"], 
                         capture_output=True, check=True)
            return True
        except (subprocess.CalledProcessError, FileNotFoundError):
            return False
    
    def run(self):
        print(self.msg("welcome"))
        print()
        
        # Check Node.js
        print(self.msg("check_node"))
        if not self.check_command("node"):
            print("✗ Node.js not found")
            sys.exit(1)
        
        result = subprocess.run(["node", "--version"], 
                              capture_output=True, text=True)
        version = result.stdout.strip().lstrip('v')
        print(self.msg("node_found", version=version))
        
        # Install Claude CLI
        print("Installing Claude Code CLI...")
        subprocess.run(["npm", "install", "-g", 
                       "@anthropic-ai/claude-code"], check=True)
        
        print()
        print(self.msg("success"))

if __name__ == "__main__":
    lang = "en"
    if len(sys.argv) > 1 and sys.argv[1].startswith("--lang="):
        lang = sys.argv[1].split("=")[1]
    
    installer = Installer(lang)
    installer.run()
'''
```

## 总结

这个方案采用**分层架构**：

1. **底层**：i18n 基础设施（独立模块，可被任何组件使用）
2. **中层**：Python/Shell 语言插件（扩展 P3.2 框架）
3. **上层**：通用项目模板（Python 项目、Shell 脚本项目）
4. **应用层**：Claude CLI 安装脚本（基于 Shell 模板 + i18n）

**优势**：
- ✅ 通用性：Python/Shell 支持可用于任何项目生成
- ✅ 可复用：i18n 系统独立，可用于所有需要多语言的场景
- ✅ 扩展性：添加新语言（Go/Rust）只需实现对应插件
- ✅ 一致性：所有语言遵循相同的插件接口
